. capture program drop tmoline
{\smallskip}
. program tmoline
  1.     args label option
  2.     local se = cond("`option'"=="scpc", e(scpc_se), e(tmo_se))
  3.     display "`label'" _col(16) "SE = " \%5.3f `se' "   ratio = " \%4.2f `se'
> /se0 _continue
  4.     if "`option'"=="" display "   pairs = " \%3.1f e(pct_ge_thres) "\%"
  5.     else display
  6. end
{\smallskip}
