. local threshold = e(threshold)
{\smallskip}

. keep fips STATE judicial_divisions_id pop60 family_less_3000 urbanB60

. use bernini_dyad.dta, clear

. generate byte selected = abs(corr) >= `threshold' \& !missing(corr)

. quietly merge m:1 id1 using `loc1', keep(match) nogenerate
{\smallskip}
. quietly merge m:1 id2 using `loc2', keep(match) nogenerate
{\smallskip}
. generate byte within150 = dist <= 150
{\smallskip}
. generate byte samestate = state1 == state2
{\smallskip}
. generate byte samejdiv  = jdiv1 == jdiv2
{\smallskip}
. foreach p in pop pov urb {\lbr}
  2.     generate d`p' = abs(`p'1 - `p'2)
  3.     quietly _pctile d`p', p(10)
  4.     generate byte close`p' = d`p' <= r(r1) if !missing(d`p')
  5. {\rbr}
{\smallskip}
. generate byte any = within150 | samestate | samejdiv | closepop | closepov | 
> closeurb

. display "criterion" _col(14) "selected pairs" _col(32) "all pairs"
criterion    selected pairs    all pairs

. foreach v in within150 samestate samejdiv closepop closepov closeurb any {\lbr}
  2.     quietly summarize `v' if selected
  3.     local sel = 100*r(mean)
  4.     quietly summarize `v'

  6. {\rbr}
